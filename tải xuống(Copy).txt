local function notify(msg, dur)
    if not Config.NotificationsEnabled then return end
    Rayfield:Notify({ Title = "Dead Hub", Content = msg, Duration = dur or 2 })
end

local function getHeadPart(char)
    if not char then return nil end
    local head = char:FindFirstChild("Head") or char:FindFirstChild("HeadPart") or char:FindFirstChild("Cabeça")
    if head then return head end
    for _, p in pairs(char:GetChildren()) do
        if p:IsA("BasePart") and string.lower(p.Name):find("head") then return p end
    end
    return nil
end

local function getRootPart(char)
    if not char then return nil end
    return char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
end

local function getHumanoid(char)
    if not char then return nil end
    return char:FindFirstChildOfClass("Humanoid")
end

local function WorldToScreen(pos)
    local v, on = Camera:WorldToViewportPoint(pos)
    return Vector2.new(v.X, v.Y), on, v.Z
end

local function getFovRadius(deg)
    local h = Camera.ViewportSize.Y
    if h <= 0 then return 200 end
    return math.max((deg / 120) * (h / 2), 20)
end

local function isPlayerBlacklisted(player)
    if Config.BlacklistMode then
        for _, name in pairs(Config.BlacklistedPlayers) do
            if player.Name == name then return true end
        end
        return false
    else
        for _, name in pairs(Config.BlacklistedPlayers) do
            if player.Name == name then return true end
        end
        return false
    end
end

local function GetVehicle(char)
    if not char or not char.Parent then return nil end
    local cars = Workspace:FindFirstChild("__cars")
    local current = char.Parent
    for _ = 1, 10 do
        if not current or current == Workspace or current == game then break end
        if (cars and current.Parent == cars) or current:FindFirstChild("VehicleConfig") or string.lower(current.Name):find("car") then
            return current
        end
        current = current.Parent
    end
    return nil
end

local function isZombie(obj)
    if not obj or not obj:IsA("Model") then return false end
    local name = obj.Name:lower()
    return name:find("__zombie") or name:find("_zombie") or name:find("zombie")
end

local function getBestTarget(targetPartMode, fovLimit, ignoreBlacklist, includeVehicles)
    local bestTarget = nil
    local bestScore = math.huge
    local cameraPos = Camera.CFrame.Position
    local cameraLook = Camera.CFrame.LookVector

    local fovCenter = Config.FOVOnMouse and UserInputService:GetMouseLocation() or (Camera.ViewportSize / 2)

    for player, data in pairs(State.ESPData) do
        if player == LocalPlayer then continue end
        if not data.character or not data.humanoid then continue end
        if data.humanoid.Health <= 0 then continue end

        if not ignoreBlacklist and isPlayerBlacklisted(player) then
            if Config.BlacklistMode then
                if not isPlayerBlacklisted(player) then continue end
            else
                continue
            end
        end

        local targetPart
        if targetPartMode == "Head" then
            targetPart = data.head or getHeadPart(data.character)
        else
            targetPart = data.rootPart or getRootPart(data.character)
        end
        if not targetPart then continue end

        local targetPos = targetPart.Position

        if Config.AimPredict and data.rootPart then
            local vel = data.rootPart.Velocity
            local vehicle = GetVehicle(data.character)
            if vehicle then
                local primary = vehicle:FindFirstChild("PrimaryPart") or vehicle:FindFirstChild("HumanoidRootPart")
                if primary then vel = primary.AssemblyLinearVelocity end
            end
            local dist = (targetPos - cameraPos).Magnitude
            local bulletSpeed = Config.BulletSpeed or 2000
            local travelTime = dist / bulletSpeed
            targetPos = targetPos + vel * travelTime * Config.PredictMultiplier
        end

        if Config.AutoPredict and data.rootPart then
            local vel = data.rootPart.Velocity
            local vehicle = GetVehicle(data.character)
            if vehicle then
                local primary = vehicle:FindFirstChild("PrimaryPart") or vehicle:FindFirstChild("HumanoidRootPart")
                if primary then vel = primary.AssemblyLinearVelocity end
            end
            local dist = (targetPos - cameraPos).Magnitude
            local bulletSpeed = 2000
            local travelTime = dist / bulletSpeed
            targetPos = targetPos + vel * travelTime
        end

        local direction = (targetPos - cameraPos).Unit
        local angle = math.deg(math.acos(math.clamp(cameraLook:Dot(direction), -1, 1)))

        if angle > fovLimit then continue end

        local screenPos, onScreen = WorldToScreen(targetPos)
        if not onScreen then continue end
        local distFromFov = (screenPos - fovCenter).Magnitude

        if distFromFov < bestScore then
            bestScore = distFromFov
            bestTarget = targetPart
        end
    end

    if Config.ESPZombie then
        for zombie, data in pairs(State.ZombieESPData) do
            if not zombie or not zombie.Parent then continue end
            local root = data.rootPart or getRootPart(zombie)
            if not root then continue end
            local targetPos = root.Position
            local dist = (targetPos - cameraPos).Magnitude
            if dist > Config.ESPZombieDistance then continue end

            local direction = (targetPos - cameraPos).Unit
            local angle = math.deg(math.acos(math.clamp(cameraLook:Dot(direction), -1, 1)))
            if angle > fovLimit then continue end

            local screenPos, onScreen = WorldToScreen(targetPos)
            if not onScreen then continue end
            local distFromFov = (screenPos - fovCenter).Magnitude

            if distFromFov < bestScore then
                bestScore = distFromFov
                bestTarget = root
            end
        end
    end

    return bestTarget
end

    local targetPos = target.Position

    local targetCF = CFrame.new(cameraPos, targetPos)
    Camera.CFrame = Camera.CFrame:lerp(targetCF, smooth)
end

local function clearESP(player)
    local data = espData[player]
    if data then
        pcall(function()
            data.box:Remove(); data.name:Remove(); data.distance:Remove(); data.hpBar:Remove(); data.hpBg:Remove(); data.inCar:Remove(); data.line:Remove(); data.tracer:Remove()
        end)
        espData[player] = nil
    end
end

local function updateCharacterParts(player, character)
    local data = espData[player]
    if not data then return end
    data.character = character
    if character then
        data.head = getHeadPart(character)
        data.rootPart = getRootPart(character)
        data.humanoid = getHumanoid(character)
    else
        data.head = nil; data.rootPart = nil; data.humanoid = nil
    end
end

local function initESP()
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then createESP(player) end
    end
    local characterMap = {}
    for _, obj in pairs(workspace:GetDescendants()) do
        if obj:IsA("Model") and obj:FindFirstChild("Humanoid") then characterMap[obj.Name] = obj end
    end
    for player, data in pairs(espData) do
        local char = characterMap[player.Name]
        if char then updateCharacterParts(player, char) end
    end
end

local function setupESPWatchers()
    local function onCharacterAdded(player, character)
        if player == LocalPlayer then return end
        updateCharacterParts(player, character)
        character:WaitForChild("HumanoidRootPart")
        task.wait(0.1)
        updateCharacterParts(player, character)
    end
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            player.CharacterAdded:Connect(function(char) onCharacterAdded(player, char) end)
        end
    end
    Players.PlayerAdded:Connect(function(player)
        if player == LocalPlayer then return end
        createESP(player)
        player.CharacterAdded:Connect(function(char) onCharacterAdded(player, char) end)
        if player.Character then updateCharacterParts(player, player.Character) end
    end)
    Players.PlayerRemoving:Connect(clearESP)
end

initESP()
setupESPWatchers()

end)

local function clearZombieESP(zombie)

        pcall(function()
            data.box:Remove(); data.name:Remove(); data.distance:Remove()
        end)

    end
end

local function initZombieESP()
    local function checkZombie(obj)
        if isZombie(obj) then
            createZombieESP(obj)
        end
    end
    for _, obj in pairs(Workspace:GetDescendants()) do
        checkZombie(obj)
    end
    Workspace.DescendantAdded:Connect(checkZombie)
    Workspace.DescendantRemoving:Connect(function(obj)
        if isZombie(obj) then clearZombieESP(obj) end
    end)
end
initZombieESP()

local function updateZombieESP()
    if not Config.ESPZombie then
        for _, data in pairs(State.ZombieESPData) do
            data.box.Visible = false; data.name.Visible = false; data.distance.Visible = false
        end
        return
    end

    local cameraPos = Camera.CFrame.Position
    for zombie, data in pairs(State.ZombieESPData) do
        if not zombie or not zombie.Parent then
            clearZombieESP(zombie)
            continue
        end
        local root = data.rootPart or getRootPart(zombie)
        if not root then
            data.box.Visible = false; data.name.Visible = false; data.distance.Visible = false
            continue
        end
        data.rootPart = root

        local dist = (root.Position - cameraPos).Magnitude
        if dist > Config.ESPZombieDistance then
            data.box.Visible = false; data.name.Visible = false; data.distance.Visible = false
            continue
        end

        local screenPos, onScreen = WorldToScreen(root.Position)
        if not onScreen or screenPos.Z < 0 then
            data.box.Visible = false; data.name.Visible = false; data.distance.Visible = false
            continue
        end

        local boxSize = Vector2.new(40, 60)
        data.box.Position = screenPos - boxSize / 2
        data.box.Size = boxSize
        data.box.Color = Color3.fromRGB(0,255,0)
        data.box.Visible = true

        data.name.Position = Vector2.new(screenPos.X, screenPos.Y - boxSize.Y / 2 - 15)
        data.name.Text = zombie.Name
        data.name.Visible = true

        data.distance.Position = Vector2.new(screenPos.X, screenPos.Y + boxSize.Y / 2 + 5)
        data.distance.Text = math.floor(dist) .. "m"
        data.distance.Visible = true
    end
end

        lines[1].To = camPos + Vector2.new(-offset - size, -offset - size)
        lines[2].From = camPos + Vector2.new(offset, -offset)
        lines[2].To = camPos + Vector2.new(offset + size, -offset - size)
        lines[3].From = camPos + Vector2.new(-offset, offset)
        lines[3].To = camPos + Vector2.new(-offset - size, offset + size)
        lines[4].From = camPos + Vector2.new(offset, offset)
        lines[4].To = camPos + Vector2.new(offset + size, offset + size)

        local flash = Drawing.new("Square")
        flash.Position = Vector2.new(0,0)
        flash.Size = Camera.ViewportSize
        flash.Color = Color3.fromRGB(255,255,255)
        flash.Transparency = 0.7
        flash.Filled = true
        flash.Visible = true
        task.wait(0.05)
        flash:Remove()

        task.wait(0.5)
        local fadeDuration = 0.3; local startFade = tick()
        while tick() - startFade < fadeDuration do
            local alpha = 1 - ((tick() - startFade) / fadeDuration)
            local currentOffset = offset + ((1 - alpha) * 10)
            for _, l in ipairs(lines) do l.Transparency = alpha end
            lines[1].From = camPos + Vector2.new(-currentOffset, -currentOffset)
            lines[1].To = camPos + Vector2.new(-currentOffset - size, -currentOffset - size)
            lines[2].From = camPos + Vector2.new(currentOffset, -currentOffset)
            lines[2].To = camPos + Vector2.new(currentOffset + size, -currentOffset - size)
            lines[3].From = camPos + Vector2.new(-currentOffset, currentOffset)
            lines[3].To = camPos + Vector2.new(-currentOffset - size, currentOffset + size)
            lines[4].From = camPos + Vector2.new(currentOffset, currentOffset)
            lines[4].To = camPos + Vector2.new(currentOffset + size, currentOffset + size)
            RunService.RenderStepped:Wait()
        end
        for _, l in ipairs(lines) do l:Remove() end
    end)

local function createTracer(startPos, endPos, color, duration)
    if not Config.BulletTracersEnabled then return end
    task.spawn(function()
        local line = Drawing.new("Line")
        line.Thickness = 2; line.Color = color or Config.TracerColor; line.Visible = true
        line.From = Vector2.new(startPos.X, startPos.Y)
        line.To = Vector2.new(endPos.X, endPos.Y)
        local startTime = tick()
        while tick() - startTime < duration do
            local alpha = 1 - ((tick() - startTime) / duration)
            line.Transparency = 1 - alpha
            RunService.RenderStepped:Wait()
        end
        line:Remove()
    end)

    if endPos then
        local p = Instance.new("Part", Workspace)
        p.Anchored = true; p.CanCollide = false; p.Material = Enum.Material.Neon
        p.Color = color or Config.TracerColor; p.Size = Vector3.new(0.5,0.5,0.5)

        emit.Texture = "rbxassetid://243660364"
        emit.Color = ColorSequence.new(color or Config.TracerColor)

        emit.Speed = NumberRange.new(5, 15)
        emit.Lifetime = NumberRange.new(0.3, 0.6)
        emit.Rate = 0
        emit:Emit(15)
        Debris:AddItem(p, 1)
    end
end

-- 🔫 BULLET PHYSICS (HOOK)
-- =============================================
local function hookBulletModule()
    if State.OldNewBullet and State.OldStep then return end
    for _, obj in pairs(getgc(true)) do
        local isBulletMod = false
        pcall(function()
            if type(obj) == "table" and obj.newbullet and obj.step then isBulletMod = true end
        end)
        if isBulletMod then
            State.OldNewBullet = obj.newbullet; State.OldStep = obj.step
            obj.newbullet = function(...)
                local args = {...}
                if Config.BulletTPEnabled then args[4] = 1e9 end
                local tag = args[1]; local startPos = args[3] and args[3].p
                if Config.BulletTracersEnabled and startPos then State.BulletStartPos[tag] = startPos end
                if args[2] == LocalPlayer then State.LastFiredTime = tick() end

                if Config.SilentAimEnabled and args[2] == LocalPlayer and args[3] then
                    local target = getBestTarget(Config.SilentAimTargetPart, Config.SilentAimFOV, true)
                    if target then
                        local startCF = args[3]
                        local dir = (target.Position - startCF.p).unit
                        args[3] = CFrame.lookAt(startCF.p, startCF.p + dir)
                    end
                end

                if args[2] == LocalPlayer and args[1] then
                    State.MyBulletTags[args[1]] = true
                    local lockedTarget = State.SilentAimTargetPlayer or State.CurrentTarget
                    if lockedTarget then State.MB_TargetTags[args[1]] = lockedTarget end
                end

                local bullet = State.OldNewBullet(table.unpack(args))

                if Config.BulletTPEnabled and args[3] and args[2] == LocalPlayer then
                    local muzzlePos = args[3].p
                    local muzzleDir = args[3].LookVector
                    local params = RaycastParams.new()
                    params.FilterDescendantsInstances = {LocalPlayer.Character, Camera}
                    params.FilterType = Enum.RaycastFilterType.Exclude
                    local result = Workspace:Raycast(muzzlePos, muzzleDir * 5000, params)
                    local impactPos = result and result.Position or (muzzlePos + muzzleDir * 500)
                    if Config.BulletTracersEnabled then createTracer(muzzlePos, impactPos, Config.TracerColor, Config.TracerDuration) end
                    if Config.HitmarkerEnabled and result and result.Instance then drawHitmarker() end
                end
                return bullet
            end
            obj.step = function(bullet, delta)
                bullet.step = (bullet.step or 0) + 1
                if Config.BulletTPEnabled and bullet.t then bullet.t = nil end

                if Config.MagicBulletEnabled and not Config.BulletTPEnabled and State.MyBulletTags[bullet.tag] and bullet.active ~= false then
                    local targetPlayer = State.MB_TargetTags[bullet.tag]
                    if targetPlayer then
                        local td = State.ESPData[targetPlayer]
                        if td then
                            local aimPart = td.head or td.rootPart
                            if aimPart and bullet.position then
                                local toTarget = (aimPart.Position - bullet.position).unit
                                local currentVel = bullet.velocity or bullet.Velocity or (bullet.dir and bullet.dir * (bullet.speed or 2000))
                                local currentDir = currentVel and currentVel.unit or toTarget
                                local speed = (currentVel and currentVel.Magnitude or bullet.speed or 2000) * 1.5
                                if bullet.gravity then bullet.gravity = Vector3.new(0,0,0) end
                                if bullet.Gravity then bullet.Gravity = Vector3.new(0,0,0) end
                                local avoidParams = RaycastParams.new()
                                avoidParams.FilterDescendantsInstances = {LocalPlayer.Character, Camera, td.character}
                                avoidParams.FilterType = Enum.RaycastFilterType.Exclude
                                local obstacleHit = Workspace:Raycast(bullet.position, currentDir * 50, avoidParams)
                                local steerDir
                                if obstacleHit and obstacleHit.Instance then
                                    local normal = obstacleHit.Normal
                                    local avoidDir = (toTarget + normal * 1.5).unit
                                    steerDir = currentDir:Lerp(avoidDir, 0.6).unit
                                else
                                    local strength = math.clamp(Config.MagicBulletStrength, 1, 10) / 10
                                    local distToTarget = (aimPart.Position - bullet.position).Magnitude
                                    local dynamicTurn = math.max(strength * 0.35, math.clamp(1 - (distToTarget / 150), 0.1, 1))
                                    steerDir = currentDir:Lerp(toTarget, dynamicTurn).unit
                                end
                                local steerVel = steerDir * speed
                                bullet.velocity = steerVel
                                if bullet.Velocity then bullet.Velocity = steerVel end
                                if bullet.dir then bullet.dir = steerDir end
                                if bullet.Direction then bullet.Direction = steerDir end
                            end
                        end
                    end
                end

                local lastPos = bullet.position
                local result = State.OldStep(bullet, delta)
                if bullet.active == false and Config.BulletTracersEnabled then
                    local muzzle = State.BulletStartPos[bullet.tag]
                    local impact = bullet.position
                    if muzzle and impact then createTracer(muzzle, impact, Config.TracerColor, Config.TracerDuration) end
                    State.BulletStartPos[bullet.tag] = nil
                end
                return result
            end
            break
        end
    end
end
spawn(function() while not State.OldNewBullet do hookBulletModule(); wait(1) end end)

local function performJump()
    local char = LocalPlayer.Character
    if not char then return false end
    local humanoid = getHumanoid(char)
    if not humanoid or not Config.JumpEnabled or humanoid.Health <= 0 then return false end
    if not Config.AllowAirJump and humanoid.FloorMaterial == Enum.Material.Air then return false end
    humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
    local rootPart = getRootPart(char)
    if rootPart then rootPart.Velocity = Vector3.new(rootPart.Velocity.X, Config.JumpForce, rootPart.Velocity.Z) end
    return true
end

local function startJumping()
    if State.JumpTask then return end
    State.HoldingJump = true
    State.JumpTask = task.spawn(function()
        while State.HoldingJump and Config.JumpEnabled do
            performJump()
            task.wait(Config.JumpDelay)
        end
        State.JumpTask = nil
    end)
end

local function stopJumping()
    State.HoldingJump = false
    if State.JumpTask then task.cancel(State.JumpTask); State.JumpTask = nil end
end

local function nukeCars()
    if not Config.NukeCarsEnabled or State.NukeCarsActive then return end
    State.NukeCarsActive = true
    local cars = Workspace:FindFirstChild("__cars")
    if not cars then State.NukeCarsActive = false return end
    local saved = {}
    for _, car in pairs(cars:GetChildren()) do
        local carParts = {}
        for _, p in pairs(car:GetDescendants()) do
            if p:IsA("BasePart") then
                table.insert(carParts, {part = p, transparency = p.Transparency, canCollide = p.CanCollide})
                p.Transparency = 1; p.CanCollide = false
            end
        end
        table.insert(saved, carParts)
    end
    notify("Cars nuked for 5 seconds", 2)
    task.wait(5)
    for _, carParts in ipairs(saved) do
        for _, data in ipairs(carParts) do
            if data.part and data.part.Parent then
                data.part.Transparency = data.transparency; data.part.CanCollide = data.canCollide
            end
        end
    end
    State.NukeCarsActive = false
    notify("Cars restored", 2)
end

local function teleportPlayer()
    if not Config.TeleportEnabled then return end
    notify("Waiting 10 seconds...", 3)
    task.wait(10)
    if not Config.TeleportEnabled then notify("Teleport cancelled!", 2) return end
    local success, err = pcall(function()
        local teleportRemote = ReplicatedStorage:FindFirstChild("RemoteFunctions")
        if teleportRemote then teleportRemote = teleportRemote:FindFirstChild("Teleport") end
        if teleportRemote then teleportRemote:InvokeServer() else error("Teleport remote not found") end
    end)
    if success then notify("Teleported!", 2) else notify("Error: " .. tostring(err), 3) end
end

local function autoFlick()
    if not Config.AutoFlickEnabled then return end
    if UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then return end
    local target = getBestTarget("Head", 30, true)
    if target then
        local randomOffset = Vector3.new(math.random(-5,5)*0.1, math.random(-5,5)*0.1, 0)
        local targetPos = target.Position + randomOffset
        local cameraPos = Camera.CFrame.Position
        local targetCF = CFrame.new(cameraPos, targetPos)
        Camera.CFrame = Camera.CFrame:lerp(targetCF, Config.AutoFlickSpeed * (0.5 + math.random() * 0.3))
    end

local function applyAntiRecoil()
    if not Config.AntiRecoilEnabled then return end
    if not UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then return end
    local recoilOffset = Vector3.new(math.random(-2,2)*0.01, -Config.AntiRecoilStrength*0.1 + math.random(-2,2)*0.01, 0)
    Camera.CFrame = Camera.CFrame * CFrame.new(recoilOffset)
end

local function applyAimAssist()
    if not Config.AimAssistEnabled then return end
    local target = getBestTarget("Head", Config.AimAssistFOV, true)
    if target then
        local targetPos = target.Position
        local cameraPos = Camera.CFrame.Position
        local direction = (targetPos - cameraPos).Unit
        local currentLook = Camera.CFrame.LookVector
        local angle = math.deg(math.acos(math.clamp(currentLook:Dot(direction), -1, 1)))
        if angle < Config.AimAssistFOV then
            local targetCF = CFrame.new(cameraPos, targetPos)
            Camera.CFrame = Camera.CFrame:lerp(targetCF, Config.AimAssistStrength)
        end
    end
end

RunService.RenderStepped:Connect(function(deltaTime)
    -- Spin
    if Config.SpinEnabled then
        local char = LocalPlayer.Character
        if char then
            local rootPart = getRootPart(char)
            if rootPart then
                State.SpinAngle = State.SpinAngle + (Config.SpinSpeed * 2 * math.pi * deltaTime)
                rootPart.CFrame = rootPart.CFrame * CFrame.Angles(0, State.SpinAngle, 0)
            end
        end
    end

    -- Float
    if Config.FloatEnabled then
        local char = LocalPlayer.Character
        if char then
            local humanoid = getHumanoid(char); local rootPart = getRootPart(char)
            if humanoid and rootPart then
                local ray = Ray.new(rootPart.Position, Vector3.new(0, -Config.FloatHeight, 0))
                local hit, pos = Workspace:FindPartOnRay(ray, char)
                if hit then
                    local targetY = pos.Y + Config.FloatHeight
                    if rootPart.Position.Y < targetY then
                        rootPart.Velocity = Vector3.new(rootPart.Velocity.X, (targetY - rootPart.Position.Y) * 0.5, rootPart.Velocity.Z)
                    end
                end
            end
        end
    end

    -- Aims
    applyAimAssist()
    if Config.AutoFlickEnabled then autoFlick() end
    if Config.TriggerbotEnabled then triggerbot() end
    if Config.AntiRecoilEnabled then applyAntiRecoil() end

    -- Rainbow
    State.RainbowHue = (State.RainbowHue + Config.RainbowSpeed * 0.001) % 1

    -- Visuals
    customizeGuns()
    updateSelfChams()
    updateGodAura()
    updateShadowClone()
    updateZombieESP()

    -- Aimlock
    if Config.AimlockEnabled then
        local shouldAim = Config.AimlockAuto or UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
        if shouldAim then
            local target = getBestTarget(Config.AimlockTargetPart, Config.AimlockFOV, true)
            if target then
                if State.IsOnCooldown then
                    -- do nothing
                elseif State.CurrentAimTarget == target then
                    State.AimTimer = State.AimTimer + deltaTime
                    if State.AimTimer >= Config.MAX_AIM_TIME then
                        State.IsOnCooldown = true; State.AimCooldownTimer = 0; State.CurrentAimTarget = nil
                    else
                        applyAim(target, Config.AimlockSmoothing)
                    end
                else
                    State.AimTimer = 0; State.CurrentAimTarget = target
                    applyAim(target, Config.AimlockSmoothing)
                end
            else
                State.AimTimer = 0; State.CurrentAimTarget = nil
            end
        end
    end

    if State.IsOnCooldown then
        State.AimCooldownTimer = State.AimCooldownTimer + deltaTime
        if State.AimCooldownTimer >= Config.AIM_COOLDOWN then
            State.IsOnCooldown = false; State.AimTimer = 0
        end
    end

    -- FOV Circle
    if Config.DrawFOV then
        fovCircle.Position = Config.FOVOnMouse and UserInputService:GetMouseLocation() or (Camera.ViewportSize / 2)
        fovCircle.Radius = getFovRadius(Config.AimlockFOV)
        fovCircle.Color = Config.AimlockEnabled and Color3.new(0,255,0) or Color3.new(255,0,0)
        fovCircle.Visible = true
    else fovCircle.Visible = false end

    -- FOV Tracer
    if Config.TracerLine then
        local target = getBestTarget(Config.AimlockTargetPart, Config.AimlockFOV, true)
        if target then
            local screenPos = Camera:WorldToViewportPoint(target.Position)
            if screenPos.Z > 0 then
                local center = Config.FOVOnMouse and UserInputService:GetMouseLocation() or (Camera.ViewportSize / 2)
                fovTracer.From = center; fovTracer.To = Vector2.new(screenPos.X, screenPos.Y); fovTracer.Visible = true
            else fovTracer.Visible = false end
        else fovTracer.Visible = false end
    else fovTracer.Visible = false end

    -- ===== PLAYER ESP (CORRIGIDO – NÃO BUGA AO VIRAR) =====
    if not Config.ESPEnabled then
        for _, data in pairs(espData) do
            data.box.Visible = false; data.name.Visible = false; data.distance.Visible = false; data.hpBar.Visible = false; data.hpBg.Visible = false; data.inCar.Visible = false; data.line.Visible = false; data.tracer.Visible = false
        end
    else
        local viewport = Camera.ViewportSize
        for player, data in pairs(espData) do
            local character = data.character
            local humanoid = data.humanoid
            if not character or not humanoid then
                data.box.Visible = false; data.name.Visible = false; data.distance.Visible = false; data.hpBar.Visible = false; data.hpBg.Visible = false; data.inCar.Visible = false; data.line.Visible = false; data.tracer.Visible = false
                continue
            end

            local isInVehicle = Lighting:FindFirstChild(player.Name) ~= nil
            local vehicle = isInVehicle and Lighting:FindFirstChild(player.Name) or nil

            local rootPart = data.rootPart
            local head = data.head
            local headPos, footPos

            if isInVehicle and vehicle then
                local primaryPart = vehicle:FindFirstChild("PrimaryPart") or vehicle:FindFirstChild("HumanoidRootPart")
                if primaryPart then
                    local pos = primaryPart.Position
                    headPos = pos + Vector3.new(0, 2.5, 0)
                    footPos = pos - Vector3.new(0, 1.5, 0)
                    rootPart = primaryPart
                    head = primaryPart
                else
                    local pos = vehicle:GetPivot().Position
                    headPos = pos + Vector3.new(0, 2.5, 0)
                    footPos = pos - Vector3.new(0, 1.5, 0)
                    rootPart = vehicle
                    head = vehicle
                end
            else
                if not rootPart or not head then
                    data.box.Visible = false; data.name.Visible = false; data.distance.Visible = false; data.hpBar.Visible = false; data.hpBg.Visible = false; data.inCar.Visible = false; data.line.Visible = false; data.tracer.Visible = false
                    continue
                end
                headPos = head.Position
                -- Pés: posição do root menos 2.5 (ajustável)

            end

            local headScreen = Camera:WorldToViewportPoint(headPos)
            local footScreen = Camera:WorldToViewportPoint(footPos)

            if height < 5 then height = 5 end
            local width = height * 0.4

            end

                data.distance.Position = Vector2.new(headScreen.X, headScreen.Y + height + 5)
                data.distance.Text = dist .. "m"
                data.distance.Visible = true
            else
                data.distance.Visible = false
            end

            if Config.ESPInCar and isInVehicle then
                data.inCar.Position = Vector2.new(headScreen.X, headScreen.Y + height + 25)
                data.inCar.Visible = true
            else
                data.inCar.Visible = false
            end

                data.hpBar.Visible = true
            else
                data.hpBar.Visible = false
                data.hpBg.Visible = false
            end

            if Config.ESPLines then

            data.line.From = Camera.ViewportSize / 2; data.line.To = center; data.line.Color = Config.ESPColor; data.line.Visible = Config.TracersEnabled

        for trap, data in pairs(State.TrapESPData) do
            if not trap or not trap.Parent then

                continue
            end
            local primary = data.primaryPart or getTrapPrimaryPart(trap)
            if not primary then
                data.box.Visible = false; data.name.Visible = false; data.distance.Visible = false
                continue
            end
            data.primaryPart = primary
            local pos = primary.Position; local screenPos, onScreen = Camera:WorldToViewportPoint(pos)
            if not onScreen or screenPos.Z < 0 then
                data.box.Visible = false; data.name.Visible = false; data.distance.Visible = false
                continue
            end
            local dist = (Camera.CFrame.Position - pos).Magnitude
            local scale = 1 / (dist + 1) * 30; local boxSize = Vector2.new(scale * 1.5, scale * 1.5); local center = Vector2.new(screenPos.X, screenPos.Y)
            data.box.Position = center - boxSize / 2; data.box.Size = boxSize; data.box.Visible = true
            data.name.Position = Vector2.new(screenPos.X, screenPos.Y - boxSize.Y / 2 - 15); data.name.Text = trap.Name; data.name.Visible = true
            data.distance.Position = Vector2.new(screenPos.X, screenPos.Y + boxSize.Y / 2 + 5); data.distance.Text = math.floor(dist) .. "m"; data.distance.Visible = Config.ESPDistance
        end
    else
        for _, data in pairs(State.TrapESPData) do data.box.Visible = false; data.name.Visible = false; data.distance.Visible = false end
    end

    -- Airdrop ESP
    if Config.ESPAirdrop then
        for airdrop, data in pairs(State.AirdropESPData) do
            if not airdrop or not airdrop.Parent then
                local d = State.AirdropESPData[airdrop]; if d then pcall(function() d.box:Remove(); d.name:Remove(); d.distance:Remove() end); State.AirdropESPData[airdrop] = nil end
                continue
            end
            local primary = data.primaryPart or getAirdropPrimaryPart(airdrop)
            if not primary then
                data.box.Visible = false; data.name.Visible = false; data.distance.Visible = false
                continue
            end
            data.primaryPart = primary
            local pos = primary.Position; local screenPos, onScreen = Camera:WorldToViewportPoint(pos)
            if not onScreen or screenPos.Z < 0 then
                data.box.Visible = false; data.name.Visible = false; data.distance.Visible = false
                continue
            end
            local dist = (Camera.CFrame.Position - pos).Magnitude
            local scale = 1 / (dist + 1) * 40; local boxSize = Vector2.new(scale * 2, scale * 2); local center = Vector2.new(screenPos.X, screenPos.Y)
            data.box.Position = center - boxSize / 2; data.box.Size = boxSize; data.box.Visible = true
            data.name.Position = Vector2.new(screenPos.X, screenPos.Y - boxSize.Y / 2 - 15); data.name.Text = "📦 " .. airdrop.Name; data.name.Visible = true
            data.distance.Position = Vector2.new(screenPos.X, screenPos.Y + boxSize.Y / 2 + 5); data.distance.Text = math.floor(dist) .. "m"; data.distance.Visible = Config.ESPDistance

    else
        for _, data in pairs(State.AirdropESPData) do data.box.Visible = false; data.name.Visible = false; data.distance.Visible = false end

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if not Config.KeybindsEnabled then return end  -- 🔥 Se as hotkeys estiverem desativadas, ignora

    local key = input.KeyCode

    -- F1 = ESP
    if key == Enum.KeyCode.F1 then
        Config.ESPEnabled = not Config.ESPEnabled
        if Config.ESPEnabled then refreshAllESP() end
        notify("ESP " .. (Config.ESPEnabled and "ON" or "OFF"), 1)
    end

    -- C = Aimlock
    if key == Enum.KeyCode.C then
        Config.AimlockEnabled = not Config.AimlockEnabled
        notify("Aimlock " .. (Config.AimlockEnabled and "ON" or "OFF"), 1)
    end

    -- V = Manuel Fly
    if key == Enum.KeyCode.V then
        Config.ManuelFlyEnabled = not Config.ManuelFlyEnabled
        notify("Fly " .. (Config.ManuelFlyEnabled and "ON" or "OFF"), 1)
    end

    -- P = Jump
    if key == Enum.KeyCode.P then
        Config.JumpEnabled = not Config.JumpEnabled
        if not Config.JumpEnabled then stopJumping() end
        notify("Jump " .. (Config.JumpEnabled and "ON" or "OFF"), 1)
    end

    -- T = TPWalk
    if key == Enum.KeyCode.T then
        Config.TPWalkEnabled = not Config.TPWalkEnabled
        notify("TPWalk " .. (Config.TPWalkEnabled and "ON" or "OFF"), 1)
    end

    -- ; = WalkFling
    if key == Enum.KeyCode.Semicolon then
        Config.WalkFlingEnabled = not Config.WalkFlingEnabled
        if Config.WalkFlingEnabled then startWalkFling() else stopWalkFling() end
        notify("WalkFling " .. (Config.WalkFlingEnabled and "ON" or "OFF"), 1)
    end

    -- G = Teleport
    if key == Enum.KeyCode.G then
        Config.TeleportEnabled = not Config.TeleportEnabled
        if Config.TeleportEnabled then
            notify("Teleport ENABLED (press G again to disable)", 2)
            task.spawn(teleportPlayer)
        else
            notify("Teleport DISABLED", 1)
        end
    end

    -- F5 = Nuke Cars
    if key == Enum.KeyCode.F5 then
        if Config.NukeCarsEnabled then
            task.spawn(nukeCars)
        end
    end

    -- X = Restaurar parede (já existia)
    if key == Enum.KeyCode.X then
        if #State.DeletedParts > 0 then
            local data = table.remove(State.DeletedParts)
            local part = data.part
            if part and part.Parent then
                pcall(function()
                    part.Transparency = data.transparency
                    part.CanCollide = data.canCollide
                    part.CastShadow = data.castShadow
                    part.Material = data.material
                    part.Anchored = data.anchored
                    part.Position = data.position
                    notify("Wall restored!", 0.5)
                end)
            end
        else
            notify("Nothing to restore!", 0.5)
        end
    end

    -- Ctrl + Clique = Delete Wall (já está no código principal)
end)
